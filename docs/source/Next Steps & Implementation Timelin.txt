// functions/src/index.ts
import * as functions from 'firebase-functions';
import * as admin from 'firebase-admin';
import OpenAI from 'openai';
import nodemailer from 'nodemailer';

admin.initializeApp();

// Email sending function
export const sendEmail = functions.https.onCall(async (data, context) => {
  if (!context.auth) {
    throw new functions.https.HttpsError('unauthenticated', 'User must be authenticated');
  }

  const { to, subject, body, attachments } = data;

  try {
    const transporter = nodemailer.createTransport({
      service: 'gmail',
      auth: {
        user: functions.config().email.user,
        pass: functions.config().email.password
      }
    });

    const mailOptions = {
      from: functions.config().email.user,
      to,
      subject,
      html: body,
      attachments
    };

    await transporter.sendMail(mailOptions);
    return { success: true };
  } catch (error) {
    console.error('Email sending failed:', error);
    throw new functions.https.HttpsError('internal', 'Failed to send email');
  }
});

// AI integration function
export const getOpenAIClient = functions.https.onCall(async (data, context) => {
  if (!context.auth) {
    throw new functions.https.HttpsError('unauthenticated', 'User must be authenticated');
  }

  const openai = new OpenAI({
    apiKey: functions.config().openai.key
  });

  return { initialized: true };
});

// Payment processing functions
export const processStripePayment = functions.https.onCall(async (data, context) => {
  if (!context.auth) {
    throw new functions.https.HttpsError('unauthenticated', 'User must be authenticated');
  }

  const { amount, currency, paymentMethodId } = data;

  try {
    const stripe = require('stripe')(functions.config().stripe.secret_key);
    
    const paymentIntent = await stripe.paymentIntents.create({
      amount: amount * 100,
      currency: currency || 'usd',
      payment_method: paymentMethodId,
      confirm: true,
      return_url: 'https://yourapp.com/return'
    });

    return {
      success: true,
      paymentIntentId: paymentIntent.id,
      status: paymentIntent.status
    };
  } catch (error) {
    console.error('Payment processing failed:', error);
    throw new functions.https.HttpsError('internal', 'Payment processing failed');
  }
});

// Campaign metrics aggregation
export const aggregateCampaignMetrics = functions.firestore
  .document('campaigns/{campaignId}')
  .onUpdate(async (change, context) => {
    const newValue = change.after.data();
    const oldValue = change.before.data();
    
    // Check if metrics were updated
    if (newValue.metrics === oldValue.metrics) {
      return;
    }

    // Update aggregated metrics in clients document
    const clientId = newValue.clientId;
    const clientRef = admin.firestore().doc(`clients/${clientId}`);
    
    // Get all campaigns for this client
    const campaignsSnapshot = await admin.firestore()
      .collection('campaigns')
      .where('clientId', '==', clientId)
      .get();

    let totalSpent = 0;
    let totalReach = 0;
    let totalEngagement = 0;
    let activeCampaigns = 0;

    campaignsSnapshot.forEach(doc => {
      const data = doc.data();
      if (data.status === 'active') activeCampaigns++;
      totalSpent += data.spent || 0;
      totalReach += data.metrics?.reach || 0;
      totalEngagement += data.metrics?.engagement || 0;
    });

    await clientRef.update({
      'stats.totalSpent': totalSpent,
      'stats.totalReach': totalReach,
      'stats.totalEngagement': totalEngagement,
      'stats.activeCampaigns': activeCampaigns,
      'stats.lastUpdated': admin.firestore.FieldValue.serverTimestamp()
    });
  });
